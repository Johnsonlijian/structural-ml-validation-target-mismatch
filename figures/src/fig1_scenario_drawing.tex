% =====================================================================
% Panel B - the two validation protocols
% =====================================================================
\begin{scope}[shift={(6.95,0)}]
  \draw[rounded corners=2pt, draw=black!25] (-0.12,-0.45) rectangle (5.12,6.62);
  \node[anchor=north west, font=\scriptsize\bfseries] at (0.0,6.50) {B. Evidence protocol};

  % random folds: test folds resample the same programmes
  \node[font=\tiny, anchor=west] at (0.05,6.05) {random $K$-fold};
  \foreach \i in {0,...,5} {
    \foreach \j in {0,...,2} {
      \pgfmathsetmacro{\c}{int(mod(\i+\j,2))}
      \ifnum\c=0 \fill[blue!35] (0.20+\i*0.38,5.25+\j*0.23) rectangle ++(0.32,0.19);
      \else \fill[orange!45] (0.20+\i*0.38,5.25+\j*0.23) rectangle ++(0.32,0.19); \fi
      \draw[draw=black!35, line width=0.2pt] (0.20+\i*0.38,5.25+\j*0.23) rectangle ++(0.32,0.19);
    }
  }
  \node[font=\tiny, text=black!60, anchor=north west, align=left] at (0.05,5.12)
    {test folds resample the same programmes;\\ the risk constraint stays inactive, so\\ the target leaves the rule unchanged};

  % source-honouring: whole programmes held out
  \node[font=\tiny, anchor=west] at (0.05,4.20) {source-honouring $K$-fold};
  \foreach \i in {0,...,5} {
    \foreach \j in {0,...,2} {
      \ifnum\i=4 \fill[orange!45] (0.20+\i*0.38,3.40+\j*0.23) rectangle ++(0.32,0.19);
      \else \fill[blue!35] (0.20+\i*0.38,3.40+\j*0.23) rectangle ++(0.32,0.19); \fi
      \draw[draw=black!35, line width=0.2pt] (0.20+\i*0.38,3.40+\j*0.23) rectangle ++(0.32,0.19);
    }
  }
  \node[font=\tiny, text=black!60, anchor=north west, align=left] at (0.05,3.27)
    {whole programmes are held out, so the\\ estimate follows the target and the\\ selected rule changes with it};

  % no run-specific numbers here: they belong to one results table and one figure
  \draw[draw=black!20] (0.05,2.45) -- (4.95,2.45);
  \node[font=\tiny, anchor=north west, align=left] at (0.05,2.33)
    {The protocol therefore changes \emph{which} rule is\\ selected and \emph{what it delivers} on unseen
     sources.\\[0.4mm]
     \textcolor{black!60}{Measured rates, denominators and coverage are reported together in
     Table~5 and Figure~5; this schematic carries no run-specific numbers.}};
\end{scope}
